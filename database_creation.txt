    CUSTOMER_ID      NUMBER(4,0)     NOT NULL,

Customer_ID number(4),
Order_date DATE,
Status Char(1),
Shipping_address varchar2(300),

CONSTRAINT PK_Order_ID PRIMARY KEY (Order_ID),
CONSTRAINT CHK_Status_Valid CHECK (Status IN ('A', 'C', 'D'))
);

order_ID NUMBER(5),
Model_no VARCHAR2(50),
Quantity Number(3),
Unit_price Number(8,2),
